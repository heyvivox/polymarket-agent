def calculate_weighted_signal_score(indicators_dict: dict) -> tuple:
    """
    Returns (weighted_score, num_healthy_indicators)
    
    Momentum (weight=2.0):
      - RSI > -1 (not sold out) → +2 points if bullish, -2 if bearish
      - MACD histogram > 0 → +2 points
      - Price rate-of-change > 0 → +1 point
    
    Volatility + Structure (weight=1.0):
      - Bollinger Band position
      - ATR direction
    
    Mean-reversion (weight=0.5):
      - Stochastic (lower weight for 5-min)
    
    Health checks:
      - RSI 20-80 range (discard if extreme)
      - MACD histogram accelerating (not decelerating)
    """
    score = 0
    health_count = 0
    
    # Momentum signals (highest weight)
    rsi = indicators_dict.get("rsi")
    if rsi and 20 < rsi < 80:  # healthy range
        health_count += 1
        if rsi > 50:
            score += 2.0
        else:
            score -= 2.0
    
    macd_hist = indicators_dict.get("macd_histogram")
    if macd_hist is not None:
        health_count += 1
        if macd_hist > 0:
            score += 2.0
        else:
            score -= 2.0
    
    # Volatility signals
    bb_pos = indicators_dict.get("bollinger_position")  # 0-1, 0.5 = middle
    if bb_pos is not None:
        health_count += 1
        if bb_pos < 0.3:  # near lower band → bullish squeeze
            score += 1.0
        elif bb_pos > 0.7:  # near upper band → bearish extension
            score -= 1.0
    
    # Mean-reversion (low weight)
    stoch_k = indicators_dict.get("stochastic_k")
    if stoch_k and 25 < stoch_k < 75:
        health_count += 0.5
        if stoch_k > 50:
            score += 0.5
        else:
            score -= 0.5
    
    return score, health_count

# In entry logic:
